+#[derive(Debug)]
+struct RotatingPolicy {
+    nodes: Vec<Arc<Node>>,
+    next: AtomicUsize,
+    allow_affinity: AtomicBool,
+    checked: Arc<Mutex<Vec<SocketAddr>>>,
+}
+
+impl RotatingPolicy {
+    fn new(nodes: Vec<Arc<Node>>, allow_affinity: bool) -> Self {
+        Self {
+            nodes,
+            next: AtomicUsize::new(0),
+            allow_affinity: AtomicBool::new(allow_affinity),
+            checked: Arc::new(Mutex::new(Vec::new())),
+        }
+    }
+}
+
+impl LoadBalancingPolicy for RotatingPolicy {
+    fn pick<'a>(
+        &'a self,
+        _routing: &'a RoutingInfo,
+        _cluster: &'a ClusterState,
+    ) -> Option<(NodeRef<'a>, Option<Shard>)> {
+        let index = self.next.fetch_add(1, Ordering::Relaxed) % self.nodes.len();
+        Some((&self.nodes[index], None))
+    }
+
+    fn fallback<'a>(
+        &'a self,
+        _routing: &'a RoutingInfo,
+        _cluster: &'a ClusterState,
+    ) -> FallbackPlan<'a> {
+        Box::new(self.nodes.iter().map(|node| (node, None)))
+    }
+
+    fn is_coordinator_preferred(
+        &self,
+        _routing: &RoutingInfo,
+        _cluster: &ClusterState,
+        coordinator: &Coordinator,
+    ) -> bool {
+        self.checked
+            .lock()
+            .unwrap()
+            .push(coordinator.connection_address());
+        self.allow_affinity.load(Ordering::Relaxed)
+    }
+
+    fn name(&self) -> String {
+        "RotatingPolicy".into()
+    }
+}

+fn handshake_rules() -> Vec<RequestRule> {
+    vec![
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Options),
+            RequestReaction::forge_response(Arc::new(|frame: RequestFrame| {
+                ResponseFrame::forged_supported(frame.params, &HashMap::new()).unwrap()
+            })),
+        ),
+        RequestRule(
+            Condition::or(
+                Condition::RequestOpcode(RequestOpcode::Startup),
+                Condition::RequestOpcode(RequestOpcode::Register),
+            ),
+            RequestReaction::forge_response(Arc::new(|frame: RequestFrame| {
+                ResponseFrame::forged_ready(frame.params)
+            })),
+        ),
+        RequestRule(
+            Condition::any([
+                Condition::RequestOpcode(RequestOpcode::Query),
+                Condition::RequestOpcode(RequestOpcode::Prepare),
+                Condition::RequestOpcode(RequestOpcode::Execute),
+            ]),
+            RequestReaction::forge().server_error(),
+        ),
+    ]
+}

+fn rows_response(frame: RequestFrame, more: bool) -> ResponseFrame {
+    rows_response_with_state(frame, more.then_some(7))
+}
+
+fn rows_response_with_state(frame: RequestFrame, state: Option<u8>) -> ResponseFrame {
+    let mut body = BytesMut::new();
+    body.put_i32(2);
+    body.put_i32(if state.is_some() { 2 } else { 0 });
+    body.put_i32(0);
+    if let Some(state) = state {
+        body.put_i32(1);
+        body.put_u8(state);
+    }
+    body.put_i32(0);
+    ResponseFrame {
+        params: frame.params.for_response(),
+        opcode: ResponseOpcode::Result,
+        body: body.freeze(),
+    }
+}
+
+fn request_paging_state(frame: &RequestFrame) -> Vec<u8> {
+    let mut body = frame.body.as_ref();
+    match frame.opcode {
+        RequestOpcode::Query => {
+            let len = body.get_i32() as usize;
+            body.advance(len);
+        }
+        RequestOpcode::Execute => {
+            let len = body.get_u16() as usize;
+            body.advance(len);
+        }
+        _ => unreachable!(),
+    }
+    QueryParameters::deserialize(&mut body)
+        .unwrap()
+        .paging_state
+        .as_bytes_slice()
+        .map(|bytes| bytes.as_ref().to_vec())
+        .unwrap_or_default()
+}
+
+fn multipage_rules(
+    query_counter: Arc<AtomicUsize>,
+    _execute_counter: Arc<AtomicUsize>,
+    pages: usize,
+    seen_states: Arc<Mutex<Vec<Vec<u8>>>>,
+) -> Vec<RequestRule> {
+    vec![
+        handshake_rules()[0].clone(),
+        handshake_rules()[1].clone(),
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Prepare),
+            RequestReaction::forge_response(Arc::new(prepared_response)),
+        ),
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Query),
+            RequestReaction::forge_response(Arc::new({
+                let seen_states = Arc::clone(&seen_states);
+                move |frame| {
+                    let request = query_counter.fetch_add(1, Ordering::SeqCst);
+                    seen_states
+                        .lock()
+                        .unwrap()
+                        .push(request_paging_state(&frame));
+                    rows_response_with_state(
+                        frame,
+                        (request + 1 < pages).then_some(11 + request as u8),
+                    )
+                }
+            })),
+        ),
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Execute),
+            RequestReaction::forge_response(Arc::new(move |frame| {
+                let state = request_paging_state(&frame);
+                seen_states.lock().unwrap().push(state.clone());
+                rows_response_with_state(frame, state.is_empty().then_some(31))
+            })),
+        ),
+    ]
+}
+
+fn prepared_response(frame: RequestFrame) -> ResponseFrame {
+    let mut body = BytesMut::new();
+    body.put_i32(4);
+    body.put_u16(1);
+    body.put_u8(42);
+    body.put_i32(0);
+    body.put_i32(0);
+    body.put_i32(0);
+    body.put_i32(0);
+    body.put_i32(0);
+    ResponseFrame {
+        params: frame.params.for_response(),
+        opcode: ResponseOpcode::Result,
+        body: body.freeze(),
+    }
+}
+
+fn paging_rules(counter: Arc<AtomicUsize>, fail_first: bool) -> Vec<RequestRule> {
+    let query_counter = Arc::clone(&counter);
+    let execute_counter = Arc::new(AtomicUsize::new(0));
+    vec![
+        handshake_rules()[0].clone(),
+        handshake_rules()[1].clone(),
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Prepare),
+            RequestReaction::forge_response(Arc::new(prepared_response)),
+        ),
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Query),
+            RequestReaction::forge_response(Arc::new(move |frame| {
+                let request = query_counter.fetch_add(1, Ordering::SeqCst);
+                if fail_first && request == 0 {
+                    ResponseFrame::forged_error(
+                        frame.params,
+                        scylla_cql::frame::response::error::DbError::Overloaded,
+                        None,
+                    )
+                    .unwrap()
+                } else {
+                    rows_response(frame, request == 0 || fail_first && request == 1)
+                }
+            })),
+        ),
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Execute),
+            RequestReaction::forge_response(Arc::new(move |frame| {
+                let request = execute_counter.fetch_add(1, Ordering::SeqCst);
+                rows_response(frame, request == 0)
+            })),
+        ),
+    ]
+}
+
+fn fail_one_request_rules(
+    attempts: Arc<AtomicUsize>,
+    successes: Arc<AtomicUsize>,
+    fail_at: usize,
+) -> Vec<RequestRule> {
+    vec![
+        handshake_rules()[0].clone(),
+        handshake_rules()[1].clone(),
+        RequestRule(
+            Condition::RequestOpcode(RequestOpcode::Query),
+            RequestReaction::forge_response(Arc::new(move |frame| {
+                let attempt = attempts.fetch_add(1, Ordering::SeqCst);
+                if attempt == fail_at {
+                    ResponseFrame::forged_error(
+                        frame.params,
+                        scylla_cql::frame::response::error::DbError::Overloaded,
+                        None,
+                    )
+                    .unwrap()
+                } else {
+                    let success = successes.fetch_add(1, Ordering::SeqCst);
+                    rows_response(frame, success == 0)
+                }
+            })),
+        ),
+    ]

+async fn session_with_two_nodes_and_profile(
+    handle: Option<scylla::client::execution_profile::ExecutionProfileHandle>,
+) -> (Session, RunningProxy) {
+    let first = SocketAddr::new(scylla_proxy::get_exclusive_local_address(), 9042);
+    let second = SocketAddr::new(scylla_proxy::get_exclusive_local_address(), 9042);
+    let proxy = Proxy::builder()
+        .with_node(
+            scylla_proxy::Node::builder()
+                .proxy_address(first)
+                .request_rules(handshake_rules())
+                .build_dry_mode(),
+        )
+        .with_node(
+            scylla_proxy::Node::builder()
+                .proxy_address(second)
+                .request_rules(handshake_rules())
+                .build_dry_mode(),
+        )
+        .build()
+        .run()
+        .await
+        .unwrap();
+
+    let mut builder = SessionBuilder::new()
+        .known_node_addr(first)
+        .known_node_addr(second);
+    if let Some(handle) = handle {
+        builder = builder.default_execution_profile_handle(handle);
+    }
+    let session = builder.build().await.unwrap();
+    assert!(session.get_cluster_state().get_nodes_info().len() >= 2);
+    (session, proxy)
+}
+
+async fn session_with_two_nodes() -> (Session, RunningProxy) {
+    session_with_two_nodes_and_profile(None).await
+}

+#[tokio::test]
+async fn manual_unprepared_affinity_is_policy_gated_and_preserves_raw_api() {
+    let (session, mut proxy) = session_with_two_nodes().await;
+    let nodes = session.get_cluster_state().get_nodes_info().to_vec();
+    let policy = Arc::new(RotatingPolicy::new(nodes.clone(), true));
+    let checked = Arc::clone(&policy.checked);
+    let handle = profile(policy);
+    let mut statement =
+        scylla::statement::unprepared::Statement::new("SELECT now() FROM system.local");
+    statement.set_execution_profile_handle(Some(handle.clone()));
+    let query_counter = Arc::new(AtomicUsize::new(0));
+    let seen_states = Arc::new(Mutex::new(Vec::new()));
+    let execute_counter = Arc::new(AtomicUsize::new(0));
+    for node in &mut proxy.running_nodes {
+        node.change_request_rules(Some(multipage_rules(
+            Arc::clone(&query_counter),
+            Arc::clone(&execute_counter),
+            3,
+            Arc::clone(&seen_states),
+        )));
+    }
+
+    let (first, continuation) = session
+        .query_single_page_with_affinity(statement.clone(), (), None)
+        .await
+        .unwrap();
+    let first_address = first.request_coordinator().connection_address();
+    assert!(!DefaultPolicy::default().is_coordinator_preferred(
+        &RoutingInfo::default(),
+        &session.get_cluster_state(),
+        first.request_coordinator(),
+    ));
+    let (second, continuation) = session
+        .query_single_page_with_affinity(statement.clone(), (), continuation)
+        .await
+        .unwrap();
+    let second_address = second.request_coordinator().connection_address();
+    let (_, continuation) = session
+        .query_single_page_with_affinity(statement.clone(), (), continuation)
+        .await
+        .unwrap();
+    assert!(continuation.is_none());
+    assert_eq!(
+        second.request_coordinator().connection_address(),
+        first_address
+    );
+    assert_eq!(&*checked.lock().unwrap(), &[first_address, second_address]);
+    let states = seen_states.lock().unwrap();
+    assert_eq!(
+        states
+            .iter()
+            .filter(|state| state.as_slice() == [11])
+            .count(),
+        1
+    );
+    assert_eq!(
+        states
+            .iter()
+            .filter(|state| state.as_slice() == [12])
+            .count(),
+        1
+    );
+
+    let raw_counter = Arc::new(AtomicUsize::new(0));
+    for node in &mut proxy.running_nodes {
+        node.change_request_rules(Some(paging_rules(Arc::clone(&raw_counter), false)));
+    }
+    for node in &mut proxy.running_nodes {
+        node.change_request_rules(Some(paging_rules(Arc::new(AtomicUsize::new(0)), false)));
+    }
+    let (_, raw_state) = session
+        .query_single_page(statement, (), PagingState::start())
+        .await
+        .unwrap();
+    assert!(matches!(
+        raw_state,
+        PagingStateResponse::HasMorePages { .. }
+    ));
+    let _ = proxy.finish().await;
+}

+            handshake_rules()[1].clone(),

+                Condition::RequestOpcode(RequestOpcode::Query),

+                        let mut body = BytesMut::new();
+                        body.extend_from_slice(uuid::Uuid::nil().as_bytes());

+#[tokio::test]
+async fn preferred_plan_preserves_order_and_suppresses_only_the_selected_target() {
+    let (session, proxy) = session_with_two_nodes().await;
+    let nodes = session.get_cluster_state().get_nodes_info().to_vec();
+    let routing = RoutingInfo::default();
+    let cluster = session.get_cluster_state();
+
+    let baseline_policy = RotatingPolicy::new(nodes.clone(), true);
+    let baseline: Vec<_> = Plan::new(&baseline_policy, &routing, &cluster)
+        .map(|(node, shard)| (node.host_id, shard))
+        .collect();
+
+    let policy = RotatingPolicy::new(nodes.clone(), true);
+    let exact: Vec<_> =
+        Plan::new_with_preferred_target(&policy, &routing, &cluster, Some((&nodes[1], Some(7))))
+            .map(|(node, shard)| (node.host_id, shard))
+            .collect();
+    let mut expected_exact = vec![(nodes[1].host_id, 7)];
+    expected_exact.extend(
+        baseline
+            .iter()
+            .copied()
+            .filter(|target| *target != (nodes[1].host_id, 7)),
+    );
+    assert_eq!(exact, expected_exact);
+
+    let policy = RotatingPolicy::new(nodes.clone(), true);
+    let materialized_exact: Vec<_> =
+        Plan::new_with_preferred_target(&policy, &routing, &cluster, Some((&nodes[1], Some(0))))
+            .map(|(node, shard)| (node.host_id, shard))
+            .collect();
+    assert_eq!(
+        materialized_exact
+            .iter()
+            .filter(|target| **target == (nodes[1].host_id, 0))
+            .count(),
+        1
+    );
+
+    let policy = RotatingPolicy::new(nodes.clone(), true);
+    let unsharded: Vec<_> =
+        Plan::new_with_preferred_target(&policy, &routing, &cluster, Some((&nodes[1], None)))
+            .map(|(node, _)| node.host_id)
+            .collect();
+    let mut expected_unsharded = vec![nodes[1].host_id];
+    expected_unsharded.extend(
+        baseline
+            .iter()
+            .map(|target| target.0)
+            .filter(|id| *id != nodes[1].host_id),
+    );
+    assert_eq!(unsharded, expected_unsharded);
+
+    let policy = RotatingPolicy::new(nodes, true);
+    let without_preference: Vec<_> =
+        Plan::new_with_preferred_target(&policy, &routing, &cluster, None)
+            .map(|(node, shard)| (node.host_id, shard))
+            .collect();
+    assert_eq!(without_preference, baseline);
+    let _ = proxy.finish().await;
+}
