+    fn infer_and_install_struct_defaults(
+        &mut self,
+        store: &mut Store,
+        name: &str,
+        const_initializers: &FieldDefaultConstInitializers,
+    ) {
+        let qualified_name = self.qualify_name(name);
+        let Some(Definition {
+            body: DefinitionBody::Struct {
+                generics, fields, ..
+            },
+            ..
+        }) = store.get_definition(qualified_name.as_str())
+        else {
+            return;
+        };
+        let generics = generics.clone();
+        let fields = fields.clone();
+
+        if self.reject_typedef_defaults(&*store, &fields) {
+            self.clear_struct_defaults(store, &qualified_name);
+            return;
+        }
+
+        let mut resolving = FxHashSet::default();
+        let sites: Vec<_> = fields
+            .iter()
+            .enumerate()
+            .filter_map(|(index, field)| {

+                    let original = *default;
+                    let default = self.materialize_field_default_consts(
+                        store,
+                        original.clone(),
+                        const_initializers,
+                        &mut resolving,
+                    );

+                    (

+                        declaration_reference,

+        }
+    }

+    fn materialize_field_default_consts(
+        &mut self,
+        store: &mut Store,
+        expression: Expression,
+        const_initializers: &FieldDefaultConstInitializers,
+        resolving: &mut FxHashSet<Symbol>,
+    ) -> Expression {
+        if let Some(path) = Self::field_default_const_path(&expression)
+            && let Some(qualified_name) = self.lookup_qualified_name(&*store, &path)
+        {
+            let qualified_name = Symbol::from(qualified_name);
+            if let Some(initializer) = const_initializers
+                .values
+                .get(qualified_name.as_str())
+                .cloned()
+            {
+                if !resolving.insert(qualified_name.clone()) {
+                    return expression;
+                }
+                let materialized = match initializer.origin {
+                    Some(origin) => self.with_file_context_mut(
+                        store,
+                        FileContext::Standard {
+                            package_id: &origin.package_id,
+                            file_id: origin.file_id,
+                            imports: &origin.imports,
+                        },
+                        |this, store| {
+                            this.materialize_field_default_consts(
+                                store,
+                                *initializer.expression,
+                                const_initializers,
+                                resolving,
+                            )
+                        },
+                    ),
+                    None => self.materialize_field_default_consts(
+                        store,
+                        *initializer.expression,
+                        const_initializers,
+                        resolving,
+                    ),
+                };
+                resolving.remove(&qualified_name);
+                return materialized;
+            }
+
+            if let Some(literal) = Self::registered_const_literal(&*store, &qualified_name) {
+                return Expression::Literal {
+                    literal,
+                    ty: Type::uninferred(),
+                    span: expression.get_span(),
+                };
+            }
+
+            if let Some(reference) =
+                self.registered_owner_aware_const(&*store, &qualified_name, expression.get_span())
+            {
+                return reference;
+            }
+        }
+
+        match expression {
+            Expression::Binary {
+                operator,
+                left,
+                right,
+                ty,
+                span,
+            } => Expression::Binary {
+                operator,
+                left: Box::new(self.materialize_field_default_consts(
+                    store,
+                    *left,
+                    const_initializers,
+                    resolving,
+                )),
+                right: Box::new(self.materialize_field_default_consts(
+                    store,
+                    *right,
+                    const_initializers,
+                    resolving,
+                )),
+                ty,
+                span,
+            },
+            Expression::Unary {
+                operator,
+                expression,
+                ty,
+                span,
+            } => Expression::Unary {
+                operator,
+                expression: Box::new(self.materialize_field_default_consts(
+                    store,
+                    *expression,
+                    const_initializers,
+                    resolving,
+                )),
+                ty,
+                span,
+            },
+            Expression::Paren {
+                expression,
+                ty,
+                span,
+            } => Expression::Paren {
+                expression: Box::new(self.materialize_field_default_consts(
+                    store,
+                    *expression,
+                    const_initializers,
+                    resolving,
+                )),
+                ty,
+                span,
+            },
+            Expression::DotAccess {
+                expression,
+                member,
+                ty,
+                span,
+                resolution,
+            } => Expression::DotAccess {
+                expression: Box::new(self.materialize_field_default_consts(
+                    store,
+                    *expression,
+                    const_initializers,
+                    resolving,
+                )),
+                member,
+                ty,
+                span,
+                resolution,
+            },
+            expression => expression,
+        }
+    }

+    fn field_default_const_path(expression: &Expression) -> Option<String> {
+        match expression {
+            Expression::Identifier { value, .. } => Some(value.to_string()),
+            Expression::DotAccess {
+                expression, member, ..
+            } => Some(format!(
+                "{}.{}",
+                Self::field_default_const_path(expression)?,
+                member
+            )),
+            _ => None,
+        }
+    }

+    fn registered_const_literal(store: &Store, qualified_name: &Symbol) -> Option<Literal> {
+        let Definition {
+            body:
+                DefinitionBody::Value {
+                    kind: ValueKind::Constant(literal),
+                    ..
+                },
+            ..
+        } = store.get_definition(qualified_name.as_str())?
+        else {
+            return None;
+        };
+        match literal {
+            // Negative integers are stored as their wrapping `u64` value, so
+            // they cannot be reconstructed as a positive literal safely here.
+            Literal::Integer { value, .. } if *value > i64::MAX as u64 => None,
+            literal => Some(literal.clone()),
+        }
+    }

+    fn registered_owner_aware_const(
+        &self,
+        store: &Store,
+        qualified_name: &Symbol,
+        span: Span,
+    ) -> Option<Expression> {
+        let package_id = store.package_for_qualified_name(qualified_name.as_str())?;
+        if package_id == self.cursor.package_id() {
+            return None;
+        }
+        let definition = store.get_definition(qualified_name.as_str())?;
+        if !definition.is_const() || !definition.visibility.is_public() {
+            return None;
+        }
+        let member = qualified_name
+            .as_str()
+            .strip_prefix(package_id)?
+            .strip_prefix('.')?;
+        if member.contains('.') {
+            return None;
+        }
+
+        Some(Expression::DotAccess {
+            expression: Box::new(Expression::Identifier {
+                value: package_id.into(),
+                ty: Type::ImportNamespace(package_id.into()),
+                span,
+                resolution: IdentifierResolution::Unresolved,
+            }),
+            member: member.into(),
+            ty: definition.ty.clone(),
+            span,
+            resolution: DotAccessResolution::PackageMember {
+                definition: Some(qualified_name.clone()),
+            },
+        })
+    }

+    fn infer_field_default_sites<K>(
+        &mut self,
+        store: &Store,
+        generics: &[Generic],
+        sites: Vec<(K, Option<Box<Expression>>, Box<Expression>, Type)>,
+    ) -> Vec<(K, Box<Expression>)> {
+        self.with_scope(|this| {
+            this.put_in_scope(generics);
+            let mut ctx = InferCtx::new(this, store);
+            sites
+                .into_iter()
+                .map(|(key, declaration_reference, default, expected_ty)| {
+                    if let Some(declaration_reference) = declaration_reference {
+                        // Preserve the referenced constant's declared type and
+                        // record its source-level use before storing the
+                        // ownership-independent materialized expression.
+                        let _ = ctx.infer_field_default(*declaration_reference, &expected_ty);
+                    }
+                    let inferred = ctx.infer_field_default(*default, &expected_ty);
+                    let mut folder = FreezeFolder::new(&ctx.env, store);
+                    let frozen = folder
+                        .freeze_items(vec![inferred])
+                        .pop()
+                        .expect("one field default was frozen");
+                    (key, Box::new(frozen))
+                })
+                .collect()
+        })
+    }
