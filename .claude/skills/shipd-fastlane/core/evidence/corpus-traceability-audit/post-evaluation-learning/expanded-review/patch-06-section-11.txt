+fn trait_object_pattern_ty(g: &AstBytecodeGen, pattern_id: PatternId) -> Option<SourceType> {
+    let pattern = g.analysis.pattern(pattern_id);
+
+    match pattern {
+        Pattern::Ident(_) | Pattern::Ctor(_) => match g.analysis.get_ident(pattern_id) {
+            Some(IdentType::Class(id, type_params)) => Some(SourceType::Class(id, type_params)),
+            Some(IdentType::Struct(id, type_params)) => Some(SourceType::Struct(id, type_params)),
+            Some(IdentType::EnumVariant(id, type_params, _)) => {
+                Some(SourceType::Enum(id, type_params))
+            }
+            Some(IdentType::Const(id)) => Some(g.sa.const_(id).ty()),
+            _ => None,
+        },
+        Pattern::LitBool(_)
+        | Pattern::LitChar(_)
+        | Pattern::LitFloat(_)
+        | Pattern::LitInt(_)
+        | Pattern::LitStr(_) => Some(g.analysis.ty(pattern_id)),
+        Pattern::Tuple(_) => Some(g.analysis.ty(pattern_id)),
