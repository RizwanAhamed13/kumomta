+fn corresponding_offset(
+    actual_file: &SyntaxNode,
+    speculative_file: &SyntaxNode,
+    speculative_offset: TextSize,
+) -> Option<TextSize> {
+    let actual_tokens = actual_file.descendants_with_tokens().filter_map(NodeOrToken::into_token);
+    let speculative_tokens =
+        speculative_file.descendants_with_tokens().filter_map(NodeOrToken::into_token);
+
+    for (actual, speculative) in actual_tokens.zip(speculative_tokens) {
+        let range = speculative.text_range();
+        if speculative_offset < range.start() || speculative_offset > range.end() {
+            continue;
+        }
+        let relative = usize::from(speculative_offset - range.start());
+        let relative = corresponding_token_offset(actual.text(), speculative.text(), relative)?;
+        return actual.text_range().start().checked_add(TextSize::try_from(relative).ok()?);
+    }
+    None
+}

+fn corresponding_token_offset(actual: &str, speculative: &str, offset: usize) -> Option<usize> {
+    if offset > speculative.len() {
+        return None;
+    }
+
+    let actual = actual.as_bytes();
+    let speculative = speculative.as_bytes();
+    let marker = COMPLETION_MARKER.as_bytes();
+    let (mut actual_offset, mut speculative_offset) = (0, 0);
+    while speculative_offset < offset {
+        if actual.get(actual_offset) == speculative.get(speculative_offset) {
+            actual_offset += 1;
+            speculative_offset += 1;
+        } else if speculative[speculative_offset..].starts_with(marker) {
+            let remaining = offset - speculative_offset;
+            if remaining <= marker.len() {
+                return Some(actual_offset);
+            }
+            speculative_offset += marker.len();
+        } else {
+            return None;
+        }
+    }
+    Some(actual_offset)
+}
